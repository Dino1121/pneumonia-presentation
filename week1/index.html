def get_group_id(filename, label):

    if label == "PNEUMONIA":
        match = re.search(
            r"(person\d+)",
            filename,
            re.IGNORECASE
        )

    elif label == "NORMAL":
        match = re.search(
            r"(IM-\d+)",
            filename,
            re.IGNORECASE
        )

    if match is None:
        raise ValueError(
            f"Group ID 추출 실패: {filename}"
        )

    return f"{label}_{match.group(1).upper()}"

sgkf = StratifiedGroupKFold(
    n_splits=10,
    shuffle=True,
    random_state=42
)

for fold, (_, fold_idx) in enumerate(
    sgkf.split(
        X=df["filepath"],
        y=df["label"],
        groups=df["group_id"]
    )
):
    df.loc[fold_idx, "fold"] = fold

df["split"] = "train"

df.loc[
    df["fold"] == 8,
    "split"
] = "val"

df.loc[
    df["fold"] == 9,
    "split"
] = "test"
